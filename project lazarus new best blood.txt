local SendDataRemote = ReplicatedStorage:FindFirstChild("ClientRemotes")
if SendDataRemote then SendDataRemote = SendDataRemote:FindFirstChild("SendData") end

local AmmoRefillEvent = nil

local function UpdateRemotes()
    if Player.Character then
        local scriptObj = Player.Character:FindFirstChild("LocalizeScript")
        if scriptObj then
            AmmoRefillEvent = scriptObj:FindFirstChild("AmmoRefill")
        end
    end
end
RunService.Heartbeat:Connect(UpdateRemotes)

local SCRIPT_ENABLED = false

-- ✅ FIXED STATUS TEXT (RAINBOW, SMALLER, NO DUPLICATES)
local activeStatusGui

local function ShowStatusText(text)
    pcall(function()
        if activeStatusGui then
            activeStatusGui:Destroy()
        end

        local sGui = Instance.new("ScreenGui")
        activeStatusGui = sGui
        sGui.ResetOnSpawn = false
        sGui.IgnoreGuiInset = true
        sGui.Parent = Player.PlayerGui

        local lbl = Instance.new("TextLabel")
        lbl.AnchorPoint = Vector2.new(0.5, 0.5)
        lbl.Position = UDim2.fromScale(0.5, 0.5)
        lbl.Size = UDim2.fromScale(0.22, 0.045)
        lbl.BackgroundTransparency = 1
        lbl.TextScaled = true
        lbl.Font = Enum.Font.GothamBold
        lbl.Text = text
        lbl.TextTransparency = 1
        lbl.Parent = sGui

        local hue = 0
        local conn
        conn = RunService.RenderStepped:Connect(function(dt)
            if not lbl.Parent then
                conn:Disconnect()
                return
            end
            hue = (hue + dt * 0.6) % 1
            lbl.TextColor3 = Color3.fromHSV(hue, 1, 1)
        end)

        TweenService:Create(lbl, TweenInfo.new(0.4), {TextTransparency = 0}):Play()

        task.delay(5, function()
            if conn then conn:Disconnect() end
            TweenService:Create(lbl, TweenInfo.new(0.7), {TextTransparency = 1}):Play()
            task.delay(0.8, function()
                if sGui then sGui:Destroy() end
            end)
        end)
    end)
end

ShowStatusText("Press B to Toggle")

-- NAMECALL HOOK
local mt = getrawmetatable(game)
local old = mt.__namecall
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if method == "FireServer" and SCRIPT_ENABLED then
        if type(args[1]) == "table" and args[1].Damage then
            args[1].Damage = 99999
            args[1].Force = 100
            args[1].GibPower = 10
        end

        if type(args[2]) == "table" and args[2].FireTime then
            args[2].FireTime = 0.02
        end

        if (self == SendDataRemote or self.Name == "FireWeapon") and AmmoRefillEvent then
            pcall(function()
                firesignal(AmmoRefillEvent.OnClientEvent, 1)
            end)
        end
    end

    return old(self, ...)
end)

setreadonly(mt, true)

UIS.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.B then
        SCRIPT_ENABLED = not SCRIPT_ENABLED
        ShowStatusText(SCRIPT_ENABLED and "UNIVERSAL MODE [ON]" or "UNIVERSAL MODE [OFF]")
    end
end)
